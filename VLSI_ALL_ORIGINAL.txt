//Behavioral implementation of 4:1 MUX using if-else
module prjt_3ub(
   input w0,
   input w1,
   input w2,
   input w3,
   input [1:0] s,
   output reg f
);

always @(*)
  if(s==2'b00)
     f=w0;
  else if(s==2'b01)
     f=w1;
  else if(s==2'b10)
     f=w2;
  else
     f=w3;
endmodule

//Testbench

module mux4_1_tb;

reg w0,w1,w2,w3;

reg [1:0] S;

wire f;

prjt_3ub uut(w0,w1,w2,w3,S,f);

initial begin

  w0=0; w1=1; w2=0; w3=1;

  S=2'b00; #10;

  S=2'b01; #10;

  S=2'b10; #10;

  S=2'b11; #10;

  $finish;

end

initial begin

  $monitor("time=%0t | S=%b | w0=%b w1=%b w2=%b w3=%b | f=%b",
       $time,S,w0,w1,w2,w3,f);

end

endmodule

// 4:1 MUX - Dataflow Modeling

module mux4to1 (w0, w1, w2, w3, S, f);
input w0, w1, w2, w3;
input [1:0] S;
output f;

assign f = S[1] ? (S[0] ? w3 : w2) :
      (S[0] ? w1 : w0);

endmodule

// 4:1 MUX - Gate-Level Modeling

module mux4to1 (w0, w1, w2, w3, S, f);
input w0, w1, w2, w3;
input [1:0] S;
output f;

wire s0, s1;
wire y0, y1, y2, y3;

not (s0, S[0]);
not (s1, S[1]);

and (y0, w0, s1, s0);
and (y1, w1, s1, S[0]);
and (y2, w2, S[1], s0);
and (y3, w3, S[1], S[0]);

or (f, y0, y1, y2, y3);

endmodule

module half_adder(
     input A,
     input B,
     output Sum,
     output Carry
     );

     assign Sum= A^B;
     assign Carry= A&B;
endmodule

module or_add(
     input C,
     input D,
     output Y
);
     assign Y=C|D;
endmodule

module full_adder(
     input A,B,Cin,
     output sum, Cout
);

wire s1,c1,c2;
half_adder HA1 (A, B, s1, c1);
half_adder HA2 (s1, Cin, sum, c2);
or_add finout (c1, c2, Cout);
endmodule

`timescale 1ns / 1ps

// Original source: VLSIPRAC3.pdf — Experiment-03: ALU WITHOUT CARRY
// Transcribed from the attached write-up. Page-wrap whitespace normalized.

// ==================== DESIGN MODULE ====================
module ALU_4bit (
    input     [3:0] A,
    input     [3:0] B,
    input     [2:0] sel,
    output reg [3:0] Y
);

always @(*)
begin
    if (sel == 3'b000)             // Addition
        Y = A + B;
    else if (sel == 3'b001)        // Subtraction
        Y = A - B;
    else if (sel == 3'b010)        // AND
        Y = A & B;
    else if (sel == 3'b011)        // OR
        Y = A | B;
    else if (sel == 3'b100)        // XOR
        Y = A ^ B;
    else if (sel == 3'b101)        // NOT
        Y = ~A;
    else if (sel == 3'b110)        // Left Shift
        Y = A << 1;
    else if (sel == 3'b111)        // Right Shift
        Y = A >> 1;
    else
        Y = 4'b0000;
end

endmodule

module alu_4bit (
    input [3:0] A,
    input [3:0] B,
    input [2:0] sel,
    output reg [3:0] Y,
    output reg carry
);

reg [4:0] temp;

always @(*) begin
    Y = 4'b0000;
    carry = 1'b0;
    temp = 5'b00000;

    case (sel)
        3'b000: begin
            temp = A + B;
            Y = temp[3:0];
            carry = temp[4];
        end

        3'b001: begin
            Y = A - B;
            carry = 1'b0;

        3'b010: Y = A & B;
        3'b011: Y = A | B;
        3'b100: Y = A ^ B;
        3'b101: Y = ~A;
        3'b110: Y = A << 1;

// Original source: practical5 VLSI srushti 5.pdf — Practical 5: JK Flip-Flop
// Transcribed from the attached write-up. Page-wrap whitespace normalized.

// ==================== DESIGN MODULE ====================
module jk_flip_flop(
    input wire clk,
    input wire rst_n,
    input wire j,
    input wire k,
    output reg q,
    output wire q_bar
);

always @(posedge clk or posedge rst_n) begin
    if (rst_n) begin
        q <= 1'b0;
    end else begin
        case ({j, k})
            2'b00: q <= q;
            2'b01: q <= 1'b0;
            2'b10: q <= 1'b1;
            2'b11: q <= ~q;
        endcase
    end
end

assign q_bar = ~q;

endmodule

// ==================== ORIGINAL TESTBENCH ====================
`timescale 1ns / 1ps

module tb_flip_flops;

reg clk;
reg rst_n;
reg j, k;
wire q_jk, q_bar_jk;

jk_flip_flop uut_jk (
    .clk(clk), .rst_n(rst_n), .j(j), .k(k),
    .q(q_jk), .q_bar(q_bar_jk)
);

always #5 clk = ~clk;

initial begin
    clk = 0; rst_n = 0; j = 0; k = 0;

    #12 rst_n = 1;

    #10 j = 1; k = 0;
    #10 j = 0; k = 1;
    #10 j = 1; k = 1;
    #20 j = 0; k = 0;

    #20 $finish;
end

endmodule

// Original source: VLSIEXP5.22.pdf — Verilog Code for T Flip-Flop
// Transcribed from the attached write-up. Page-wrap whitespace normalized.

// ==================== DESIGN MODULE ====================
module t_flip_flop (
    input wire clk, // Clock signal
    input wire rst_n, // Asynchronous active-low reset
    input wire t, // Toggle input
    output reg q, // Flip-flop output
    output wire q_bar // Inverted output
);

always @(posedge clk or negedge rst_n) begin
    if (!rst_n) begin
        q <= 1'b0;
    end else begin
        if (t)
            q <= ~q; // Toggle when T = 1
        else
            q <= q; // Hold state when T = 0
    end
end

assign q_bar = ~q;

endmodule

module t_flipflop(

    input wire rst_n,
    input wire t,
    output reg q,
    output wire q_bar,
    output reg clk_div

        clk_div <= 1'b0;

    end

    if (!rst_n) begin
        q <= 1'b0;
    end
    else if (t) begin
        q <= ~q;
    end
end

module up_down_counter_3bit (
     input wire clk,
     input wire rst_n,
     input wire up_down,
     output reg [2:0] count,
     output wire clk_div
);

     reg [25:0] counter;
     reg clk_div_reg;

     always @(posedge clk or negedge rst_n) begin
       if (!rst_n) begin
           counter <= 26'd0;
           clk_div_reg <= 1'b0;
       end
       else if (counter == 26'd49_999_999) begin
           counter <= 26'd0;
           clk_div_reg <= ~clk_div_reg;
       end
       else begin
           counter <= counter + 1'b1;
       end
     end

     assign clk_div = clk_div_reg;

     always @(posedge clk_div or negedge rst_n) begin
       if (!rst_n) begin
           count <= 3'b000;
       end
       else begin
           if (up_down)
             count <= count + 1'b1;
           else
             count <= count - 1'b1;
       end
     end

endmodule

module tb_up_down_counter_3bit;

// ==================== DESIGN MODULE ====================
module siso_4bit (
    input wire clk,
    input wire rst,
    input wire serial_in,
    output wire serial_out
);

reg [3:0] shift_reg;

always @(posedge clk or posedge rst) begin
    if (rst)
        shift_reg <= 4'b0000;
    else
        shift_reg <= {shift_reg[2:0], serial_in};
end

assign serial_out = shift_reg[3];

endmodule

// Original source: VLSI 6b.pdf — 4-bit shift register with SIPO and PISO modes
// Transcribed from the attached write-up. Page-wrap whitespace normalized.

// ==================== DESIGN MODULE ====================
module sipo_reg (
    input wire clk,
    input wire rst,
    input wire mode,          // 0 = SIPO, 1 = PISO
    input wire load,          // Load parallel data in PISO mode
    input wire serial_in,
    input wire [3:0] parallel_in,
    output reg [3:0] parallel_out,
    output wire serial_out
);

reg [3:0] shift_reg;

always @(posedge clk or posedge rst) begin
    if (rst) begin
        shift_reg <= 4'b0000;
        parallel_out <= 4'b0000;
    end
    else begin
        if (mode == 1'b0) begin
            // SIPO operation
            shift_reg <= {shift_reg[2:0], serial_in};
            parallel_out <= {shift_reg[2:0], serial_in};
        end
        else begin
            // PISO operation
            if (load)
                shift_reg <= parallel_in;
            else
                shift_reg <= {1'b0, shift_reg[3:1]};
        end
    end
end

// Serial output during PISO mode
assign serial_out = shift_reg[0];

endmodule

// ==================== ORIGINAL TESTBENCH ====================
`timescale 1ns / 1ps

module sipo_reg_tb;

reg clk;
reg rst;
reg mode;
reg load;
reg serial_in;
reg [3:0] parallel_in;

wire [3:0] parallel_out;
wire serial_out;

// Instantiate the design
sipo_reg DUT (
    .clk(clk),
    .rst(rst),
    .mode(mode),
    .load(load),
    .serial_in(serial_in),
    .parallel_in(parallel_in),
    .parallel_out(parallel_out),
    .serial_out(serial_out)
);

// Clock generation: 10 ns period
always #5 clk = ~clk;

initial begin
    clk = 0;
    rst = 1;
    mode = 0;
    load = 0;
    serial_in = 0;
    parallel_in = 4'b0000;

    $monitor("Time=%0t | mode=%b | load=%b | serial_in=%b | parallel_in=%b | parallel_out=%b | serial_out=%b",
             $time, mode, load, serial_in, parallel_in,
             parallel_out, serial_out);

    // Apply reset
    #12 rst = 0;

    // --------------------------------
    // Test 1: SIPO mode (mode = 0)
    // Send serial bits: 1, 0, 1, 1
    // --------------------------------
    mode = 0;

    @(negedge clk) serial_in = 1;
    @(negedge clk) serial_in = 0;
    @(negedge clk) serial_in = 1;
    @(negedge clk) serial_in = 1;

    @(negedge clk) serial_in = 0;

    // --------------------------------
    // Test 2: PISO mode (mode = 1)
    // Load parallel data 1011
    // --------------------------------
    mode = 1;
    parallel_in = 4'b1011;
    load = 1;

    @(negedge clk);
    load = 0;

    // Shift out all 4 bits
    repeat (4) @(negedge clk);

    #10;
    $finish;
end

endmodule
